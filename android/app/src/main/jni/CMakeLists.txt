cmake_minimum_required(VERSION 3.13)
project(cppintegration)

set(CMAKE_CXX_STANDARD 17)

add_library(cppintegration SHARED
    NativeAdder.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/generated/jni/NativeAdderModule.cpp // Generated by Codegen
    ${CMAKE_CURRENT_SOURCE_DIR}/generated/jni/NativeAdderModule.h // Generated by Codegen
)

find_package(ReactAndroid REQUIRED CONFIG)
find_package(fbjni REQUIRED CONFIG)

target_link_libraries(cppintegration
    ReactAndroid::jsi
    ReactAndroid::react_nativemodule_core
    fbjni::fbjni
    android
)
